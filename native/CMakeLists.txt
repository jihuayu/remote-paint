cmake_minimum_required(VERSION 3.24)
project(remote_paint_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT APPLE)
  message(FATAL_ERROR "The standalone native helper currently targets macOS only")
endif()
if(NOT DEFINED SKIA_ROOT)
  message(FATAL_ERROR "Pass -DSKIA_ROOT=/path/to/pinned/skia")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(HB REQUIRED IMPORTED_TARGET harfbuzz-subset)
pkg_check_modules(PNG REQUIRED IMPORTED_TARGET libpng)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/third_party" "${CMAKE_CURRENT_BINARY_DIR}/include/components")
file(CREATE_LINK "${SKIA_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/include/third_party/skia" SYMBOLIC)
file(CREATE_LINK "${CMAKE_CURRENT_SOURCE_DIR}/chromium" "${CMAKE_CURRENT_BINARY_DIR}/include/components/remote_paint" SYMBOLIC)
add_library(serializer STATIC chromium/serializer.cc chromium/font_subset.cc)
target_compile_options(serializer PUBLIC -fno-rtti)
target_include_directories(serializer PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/include" "${SKIA_ROOT}")
target_link_libraries(serializer PUBLIC "${SKIA_ROOT}/out/RemotePaint/libskia.a" PkgConfig::HB PkgConfig::PNG z
  "-framework ApplicationServices" "-framework CoreText" "-framework CoreGraphics" "-framework Foundation")
add_executable(skp_subset subset_main.cc)
target_link_libraries(skp_subset PRIVATE serializer)
